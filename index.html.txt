  <script>
    const noBtn = document.getElementById('no');
    const yesBtn = document.getElementById('yes');
    const modal = document.getElementById('modal');
    const closeBtn = document.getElementById('close');

    function rand(min, max){ return Math.random() * (max - min) + min; }

    // Warp the NO button anywhere in the viewport (fully visible)
    function warpNoButton(){
      const padding = 12;
      const r = noBtn.getBoundingClientRect();

      const maxX = Math.max(padding, window.innerWidth  - r.width  - padding);
      const maxY = Math.max(padding, window.innerHeight - r.height - padding);

      const x = rand(padding, maxX);
      const y = rand(padding, maxY);

      noBtn.style.left = x + 'px';
      noBtn.style.top  = y + 'px';
      noBtn.style.transform = `rotate(${rand(-10,10)}deg) scale(${rand(0.98,1.06)})`;
      noBtn.style.transition = 'left 120ms ease, top 120ms ease, transform 120ms ease';
    }

    // Extra: "near miss" detector so it warps before you click it
    function proximityWarp(e){
      const r = noBtn.getBoundingClientRect();
      const d = Math.hypot(
        e.clientX - (r.left + r.width/2),
        e.clientY - (r.top  + r.height/2)
      );
      if(d < 160) warpNoButton();
    }

    // Make it slippery on desktop
    noBtn.addEventListener('mouseenter', warpNoButton);
    document.addEventListener('mousemove', proximityWarp);

    // Make it slippery on mobile
    noBtn.addEventListener('touchstart', (e)=>{
      e.preventDefault();
      warpNoButton();
    }, {passive:false});

    // Also warp if someone somehow clicks it
    noBtn.addEventListener('click', (e)=>{
      e.preventDefault();
      warpNoButton();
    });

    // Keep it in bounds after rotate/resize
    window.addEventListener('resize', ()=>{
      warpNoButton();
    });

    // Hearts confetti
    function popHearts(){
      for(let i=0;i<18;i++){
        const h = document.createElement('div');
        h.className = 'heart';
        h.style.left = rand(10, 90) + 'vw';
        h.style.top  = rand(70, 92) + 'vh';
        h.style.color = (Math.random() > 0.5) ? '#ff4da6' : '#4deeea';
        h.style.animationDuration = rand(1.3, 2.1) + 's';
        document.body.appendChild(h);
        setTimeout(()=>h.remove(), 2300);
      }
    }

    // Shooting stars on YES click
    function shootStars(count = 7){
      for(let i=0;i<count;i++){
        const s = document.createElement('div');
        s.className = 'shootStar';
        s.style.left = rand(-10, 60) + 'vw';
        s.style.top  = rand(-10, 30) + 'vh';
        s.style.animationDelay = (i * 120) + 'ms';
        document.body.appendChild(s);
        setTimeout(()=>s.remove(), 1600);
      }
    }

    // YES action
    yesBtn.addEventListener('click', ()=>{
      popHearts();
      shootStars(7);
      modal.classList.add('show');
      modal.setAttribute('aria-hidden', 'false');
    });

    closeBtn.addEventListener('click', ()=>{
      modal.classList.remove('show');
      modal.setAttribute('aria-hidden', 'true');
    });

    modal.addEventListener('click', (e)=>{
      if(e.target === modal) closeBtn.click();
    });

    // Start position
    warpNoButton();
  </script>
